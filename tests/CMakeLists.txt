file(GLOB TEST_SOURCES ${CMAKE_CURRENT_SOURCE_DIR}/*.cpp)
foreach(testfile ${TEST_SOURCES})
    file(RELATIVE_PATH filename ${CMAKE_CURRENT_SOURCE_DIR} ${testfile})
    string(REPLACE ".cpp" "" file ${filename})
    add_executable(${file} ${testfile})
	target_link_libraries(${file} ${LINK_TEST_FLAGS})
	add_test(NAME ${file} COMMAND ${file})
endforeach(testfile ${TEST_SOURCES})

if(USE_CUDA)
    add_executable(cuda_list list.cu)
    target_compile_options(cuda_list PUBLIC ${COMPILE_TEST_FLAGS})
    target_link_libraries(cuda_list ${LINK_TEST_FLAGS})
    add_test(AllTests cuda_list)
endif()

# all `*.cu` tests, built via `cuda_tests`, ctest label `cuda`
if(CMAKE_CUDA_COMPILER)
    add_custom_target(cuda_tests)
    file(GLOB CUDA_TEST_SOURCES CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/*.cu)
    foreach(testfile ${CUDA_TEST_SOURCES})
        get_filename_component(file ${testfile} NAME_WE)
        add_executable(${file} ${testfile})
        target_compile_options(${file} PUBLIC -w -DUSE_CUDA)
        target_link_libraries(${file} GTest::gtest Threads::Threads)
        if(LINKER_SUPPORTS_ATOMIC)
            target_link_libraries(${file} atomic)
        endif()
        add_dependencies(cuda_tests ${file})
        add_test(NAME ${file} COMMAND ${file})
        set_tests_properties(${file} PROPERTIES LABELS cuda)
    endforeach()
endif()
