file(GLOB TEST_SOURCES ${CMAKE_CURRENT_SOURCE_DIR}/*.cpp)

foreach(testfile ${TEST_SOURCES})
    file(RELATIVE_PATH filename ${CMAKE_CURRENT_SOURCE_DIR} ${testfile})
    string(REPLACE ".cpp" "" file ${filename})
    add_executable(mceliece_test_pollard_${file} ${testfile})
    target_compile_options(mceliece_test_pollard_${file} PUBLIC ${COMPILE_TEST_FLAGS})
    target_link_libraries(mceliece_test_pollard_${file} ${LINK_TEST_FLAGS} ${TCMALLOC_LIBRARIES})
endforeach(testfile ${TEST_SOURCES})

if(CMAKE_CUDA_COMPILER)
    file(GLOB TEST_CUDA_SOURCES ${CMAKE_CURRENT_SOURCE_DIR}/*.cu)
    foreach(testfile ${TEST_CUDA_SOURCES})
        file(RELATIVE_PATH filename ${CMAKE_CURRENT_SOURCE_DIR} ${testfile})
        string(REPLACE ".cu" "" file ${filename})
        add_executable(cu_mceliece_test_pollard_${file} ${testfile})
        target_compile_options(cu_mceliece_test_pollard_${file} PUBLIC ${COMPILE_TEST_FLAGS})
        target_link_libraries(cu_mceliece_test_pollard_${file} ${LINK_TEST_FLAGS} )
    endforeach(testfile ${TEST_CUDA_SOURCES})
endif()
